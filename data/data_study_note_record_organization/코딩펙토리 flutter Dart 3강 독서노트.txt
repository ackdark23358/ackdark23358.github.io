final List<Map<String, String>> people = [
    {
      'name': '지수',
      'group': '블랙핑크',
    },
    {
      'name': '로제',
      'group': '블랙핑크',
    },
    {
      'name': 'RM',
      'group': 'BTS',
    },
    {
      'name': '뷔',
      'group': 'BTS',
    },
  ];

  //맵은 자유도가 높다.

  //name이 들어있는지 group이 들어있는지 알수가 없다.
  //위의 맵구조처럼 json 형태로 데이터가 들어온다.

  print(people);

  //! 값들이 무조건 존재한다.
  //조건하게 매핑을 해서 출력을 해보자.
  final parsedPeople = people
      .map(
        (x) => Person(
          name: x['name']!,
          group: x['group']!,
        ),
      )
      .toList();

  print(parsedPeople);

  print('\n');
  print('\n');

  for (Person person in parsedPeople) {
    print(person.name);
    print(person.group);
  }

  final bts = parsedPeople.where(
    (x) => x.group == 'BTS',
  );

  print('\n');
  print('\n');
  print('\n');

  print(bts);

  //다이렉트로 파싱하고 검색 무한대로 계속 할 수 있다. 펑셔널 프로그밍 장점
  //체이닝을 할 수 있다.
  //쉽게 새로운 형태로 만들 수 있다.
  
  // oop보다 쉽다.
  
  final result = people
      .map(
        (x) => Person(
          name: x['name']!,
          group: x['group']!,
        ),
      )
      .where((x) => x.group == 'BTS')
      .fold<int>(
        0,
        (prev, next) => prev + next.name.length,
      );
  print(result);
}

class Person {
  final String name;
  final String group;

  Person({
    required this.name,
    required this.group,
  });

  @override
  String toString() {
    return 'Person(name:$name, group:$group)';
  }
}
