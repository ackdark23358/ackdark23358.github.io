class Idol {
  
  final String name;
  final List<String> members;
  
  
  const Idol(this.name, this.members);
  
  
//   Idol(String name, List<String> members)
//     : this.name = name,
//     this.members = members;

  //포지셔널 파라미터 (순서 상관 o)
  //네임드 파라미터 (순서상관 X)
  
  Idol.fromList(List values)
      : this.members = values[0],
        this.name = values[1];

  
  
  void sayHello(){
    print('안녕하세요 ${this.name} 입니다.');
  }
  void introduce() {
    print('저희 맴버는 ${this.members} 가 있습니다.');
  }
}

// 상속을 받으면 부모클래스의 모두ㅡㄴ 속성을
// 자식클래스가 부여받는다.


class Idol{
  //이름
  String name;
  //맴버 숫자
  int memberCount;
  
  Idol({
    required this.name,
    required this.memberCount,
  });
  
  void sayName(){
    print('저는 ${this.name} 입니다.');
    
  }
  
  void sayMemberCount(){
    print('${this.name}은 ${this.memberCount}명의 맴버가 있습니다.');
  }
  
  
}

class BoyGroup extends Idol{
  BoyGroup(
    String name,
    int membersCount,
  ): super(name: name, memberCount: membersCount);
  
  sayHello(){
    print("hello boyGroup입니다.");
  }
}

class GirlGroup extends Idol{
  GirlGroup(
    String name,
    int memberCount,
  ): super(name: name, memberCount: memberCount);
  
  sayHello(){
    print('hello girlGroup입니다.');
  }
}

class TimesTwo {
  final int number;

  TimesTwo(
    this.number,
  );

  int calculate() {
    return number * 2;
  }
}

class TimeFour extends TimesTwo {
  TimeFour(
    int number,
  ) : super(number);
  
  @override
  int calculate(){
    return super.number * 4;
  }
}
