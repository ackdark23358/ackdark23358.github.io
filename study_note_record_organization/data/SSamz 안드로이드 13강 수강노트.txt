ActivityResultContracts.StartActivityForResult()){

}

intent.putExtra("data2", 10)
requestLauncher.launch(intent)
}
}
}

package com.example.chapter13.ActivityLifeCycle
import androidx.appcompat.app.AppCompatActivity
import android.os.Bundle
import android.os.PersistableBundle
import com.example.chapter13.databinding.ActivityLifecyclePraBinding
class LifecyclePra : AppCompatActivity() {
var count = 0
lateinit var binding: ActivityLifecyclePraBinding
override fun onCreate(savedInstanceState: Bundle?) {
super.onCreate(savedInstanceState)
//xml 파인딩
binding = ActivityLifecyclePraBinding.inflate(layoutInflater)
//버튼 클릭하면 count 증가 텍스트뷰에 변경되는값 지정 출력
binding.countButton.setOnClickListener {
count ++
binding.resultView.text = "${count}"
}
setContentView(binding.root)
}
//데이터 손실을 방지하기위해 데이터 저장
override fun onSaveInstanceState(outState: Bundle) {
super.onSaveInstanceState(outState)
outState.putInt("count",count)
}
//데이터 출력
override fun onRestoreInstanceState(savedInstanceState: Bundle) {
super.onRestoreInstanceState(savedInstanceState)
val data = savedInstanceState.getInt("count")
binding.resultView.text = "$data"
}
}

package com.example.chapter13.Corutine
import androidx.appcompat.app.AppCompatActivity
import android.os.Bundle
import com.example.chapter13.R
import com.example.chapter13.databinding.ActivityCorutinePraBinding
import kotlinx.coroutines.*
import kotlinx.coroutines.channels.Channel
import kotlinx.coroutines.channels.consumeEach
import kotlin.system.measureNanoTime
class CorutinePra : AppCompatActivity() {
override fun onCreate(savedInstanceState: Bundle?) {
super.onCreate(savedInstanceState)
//데이터 바인딩 준비
val binding = ActivityCorutinePraBinding.inflate(layoutInflater)
setContentView(binding.root)
binding.button.setOnClickListener {
val channel = Channel<Int>()
//코루틴으로 비동기 처리 연산 코드 작성
val scope = CoroutineScope(Dispatchers.Default + Job())
scope.launch {
var sum = 0L
//5초내로 응답가능하게 해야하므로 나노 밀리 세컨드 사용하자
var time = measureNanoTime {
for(i in 1..2_000_000_000){
sum += i
}
}
//코루틴에서 비동기처리로 연산을 하면서 계산되는 값을 메은쓰레드로 넘겨주자
channel.send(sum.toInt())
}
//스코프 생성 하고 런치(구동) 해서 텍스트 뷰에 찍어주자
val mainScope = GlobalScope.launch(Dispatchers.Main){
channel.consumeEach{
binding.textView.text = "sum : $it"
}
}
}
}
}
