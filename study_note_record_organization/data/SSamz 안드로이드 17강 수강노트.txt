"_id integer primary key autoincrement," +
"name not null," +
"phone)")

arrayOf("fifo23", "01000011"))
db.execSQL("insert into USER_TB (name, phone) values (?,?)",
arrayOf("kim", "010000022"))
}catch (e: Exception){ }
// 커서객체에 쿼리를 참고하게 해서 row 값 한줄씩 참조해갈때 fifo23으로 된 데이터 에서
// name과 phone을 뽑아서 로그에 출력하는 코드
binding.databaseBtn.setOnClickListener {
val cursor = db.rawQuery("select * from USER_TB", null)
//로우를 하나씩 지칭
while (cursor.moveToNext()){
val name = cursor.getString(1)
val phone = cursor.getString(2)
Log.d("fifo23", "$name, $phone")
}
}
}
}

package com.example.chapter17.Database
import android.content.Context
import androidx.appcompat.app.AppCompatActivity
import android.os.Bundle
import android.util.Log
import com.example.chapter17.databinding.ActivityPreferencePraBinding
class PreferencePra : AppCompatActivity() {
override fun onCreate(savedInstanceState: Bundle?) {
super.onCreate(savedInstanceState)
val binding = ActivityPreferencePraBinding.inflate(layoutInflater)
setContentView(binding.root)
//공유 저장공간을 얻어와서 변수에 저장
val sharedPref = getSharedPreferences("my_prefs", Context.MODE_PRIVATE)
//데이터 입력
binding.preferenceSetBtn.setOnClickListener {
sharedPref.edit().run {
putString("data1", "hello")
putInt("data2", 10)
commit()
}
}
//데이터 조회
binding.preferenceGetBtn.setOnClickListener {
val data1 = sharedPref.getString("data1","world")
val data2 = sharedPref.getInt("data2",10)
Log.d("fifo23","data1 : $data1, data2 : $data2")
}
}
}
