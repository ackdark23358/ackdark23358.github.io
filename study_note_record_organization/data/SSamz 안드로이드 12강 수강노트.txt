import com.example.chpater12.R
import com.example.chpater12.databinding.ActivityTablayoutPraBinding
import com.google.android.material.tabs.TabLayout
import kotlinx.android.synthetic.main.pager_tab.*
class TablayoutPra : AppCompatActivity() {
override fun onCreate(savedInstanceState: Bundle?) {
super.onCreate(savedInstanceState)
val binding = ActivityTablayoutPraBinding.inflate(layoutInflater)
setContentView(binding.root)
supportFragmentManager.beginTransaction().add(R.id.tabContent, FragmentOne()).commit()
binding.tabs.addOnTabSelectedListener(object : TabLayout.OnTabSelectedListener {
override fun onTabReselected(tab: TabLayout.Tab?) {
}
override fun onTabSelected(tab: TabLayout.Tab?) {
val transaction = supportFragmentManager.beginTransaction()
when(tab?.text){
"Tab1" -> transaction.replace(R.id.tabContent, FragmentOne())
"Tab2" -> transaction.replace(R.id.tabContent, FragmentTwo())
"Tab3" -> transaction.replace(R.id.tabContent, FragmentThree())
}
transaction.commit()
}
override fun onTabUnselected(tab: TabLayout.Tab?) {
}
})
}
}

package com.example.chapter12_meterial
import android.content.Context
import android.graphics.Color
import android.graphics.Rect
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.core.view.ViewCompat
import androidx.recyclerview.widget.RecyclerView
import com.example.chapter12_meterial.databinding.ItemRecylerviewBinding
//뷰홀더 클레스 정의 리사이클러뷰 사용할거기 떄문에 리사이클러뷰 홀더 사용 인자값으로 바인딩 아이템 데이타
class MyViewHolder(val binding: ItemRecylerviewBinding): RecyclerView.ViewHolder(binding.root)
//어뎁터 클레스 정의 데이타는 list 형식으로 문자열값
class MyAdapter(val datas: MutableList<String>) : RecyclerView.Adapter<RecyclerView.ViewHolder>() {
override fun getItemCount(): Int {
return datas.size
}
override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder
= MyViewHolder(ItemRecylerviewBinding.inflate(LayoutInflater.from(parent.context), parent, false))
override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
val binding = (holder as MyViewHolder).binding
binding.itemData.text=datas[position]
}
}
//데코레이션 작업하기 위한 클레스 정의 컨텍스트를 받고 리사이클러뷰를 구성할대에서에 대한 데코레이션작업
class Mydecoration(val context: Context) : RecyclerView.ItemDecoration() {
override fun getItemOffsets(
outRect: Rect,
view: View,
parent: RecyclerView,
state: RecyclerView.State
) {
super.getItemOffsets(outRect, view, parent, state)
val index = parent.getChildAdapterPosition(view) +1
//3개씩 묶기
if (index % 3 == 0 ) //left, top,right, bottom
outRect.set(10,10,10,60)
else
outRect.set(10,10,10,10)
//16진수로 배경색 컬러지정
view.setBackgroundColor(Color.parseColor("#28a0ff"))
//엘리베이션 효과 적용
ViewCompat.setElevation(view, 20.0f)
}
}

package com.example.chapter12_meterial
import androidx.appcompat.app.AppCompatActivity
import android.os.Bundle
import android.util.Log
import android.view.Menu
import android.view.MenuItem
import androidx.appcompat.app.ActionBarDrawerToggle
import androidx.fragment.app.Fragment
import androidx.fragment.app.FragmentActivity
import androidx.viewpager2.adapter.FragmentStateAdapter
import com.example.chapter12_meterial.databinding.ActivityMainBinding
import com.google.android.material.tabs.TabLayoutMediator
class MainActivity : AppCompatActivity() {
//토글버튼 인스턴스화 될때 사용할라고 선언해놓고
lateinit var toggle: ActionBarDrawerToggle
//어뎁터 하나 만들어놓고
class MyFragmentPagerAdapter(activity: FragmentActivity): FragmentStateAdapter(activity){
var fragments: List<Fragment>
init {
fragments= listOf(meterialFragmentOne(), meterialFragmentTwo(), meterialFragmentThree())
}
override fun getItemCount(): Int = fragments.size
override fun createFragment(position: Int): Fragment = fragments[position]
}
override fun onCreate(savedInstanceState: Bundle?) {
super.onCreate(savedInstanceState)
val binding = ActivityMainBinding.inflate(layoutInflater)
setContentView(binding.root)
//no Action바 해놓고 메인xml 구성되있는 엑션바 장착
setSupportActionBar(binding.meterialToolbar)
//토글버튼 인스턴트화 엑션바에 토글버튼 그리고 인스턴스화 시킨다.
toggle = ActionBarDrawerToggle(this, binding.meterialDrawer, R.string.meterial_drawer_opened,
R.string.meterial_drawer_closed)
supportActionBar?.setDisplayHomeAsUpEnabled(true)
toggle.syncState()
//어뎁터 생성
val adapter = MyFragmentPagerAdapter(this)
//페이저에 어뎁터 장착
binding.meterialViewpager.adapter = adapter
//텝과 페이저 연동
TabLayoutMediator(binding.meterialTabs, binding.meterialViewpager){
tab, position ->
tab.text = "Tab${(position + 1)}"
}.attach()
}
// 상당 옵션메뉴 구성 하기위해 호출되는 메소드 오버라이딩
// menu 폴더의 main_menu 화면에 인플레이션
override fun onCreateOptionsMenu(menu: Menu?): Boolean {
menuInflater.inflate(R.menu.menu_main, menu)
return super.onCreateOptionsMenu(menu)
}
// 상단 옵션메뉴 클릭했을때 호출되는 메소드 오버라이딩
override fun onOptionsItemSelected(item: MenuItem): Boolean {
//이벤트가 토글버튼에서 제공된거라면
if(toggle.onOptionsItemSelected(item)){
Log.d("fifo23", "토글버튼에서 아이템이 선택되었다.")
return true
}
return super.onOptionsItemSelected(item)
}
}
