import numpy as np

def total_and_mean():
    with open('sample.txt', 'r') as f:
        lines = f.readlines()
        print("파일에서 읽어온 줄들:", lines) # 어떤 내용이 들어왔는지 확인해보세요!

        data = np.array(split_lines)

        with open('result.txt', 'w', encoding='utf-8') as f:
            f.write("총합: ")
            f.write(str(total))
            f.write('\n')
            f.write("평균값: ")
            f.write(str(average))

import numpy as np

class Calculator():
    def __init__(self, data):
        self.data = data
        self.total = 0
        self.average = 0.0

    def sum(self):
        for i in range(len(self.data)):
            self.total += self.data[i]

        return self.total
    def avg(self):
        arr_data = np.array(self.data)
        self.average = float(np.mean(arr_data))

        return self.average

def DashInsert(data_str_input): # 함수의 매개변수 이름을 data_str_input으로 변경하여 문자열임을 명확히 합니다.
    # 1. 입력받은 문자열의 각 문자를 정수형 숫자로 변환하여 리스트에 저장합니다.
    # 예: "4546793" -> [4, 5, 4, 6, 7, 9, 3]
    num_list = [int(digit) for digit in data_str_input]

    # 결과를 저장할 리스트를 초기화합니다. 첫 번째 숫자는 무조건 들어가야 합니다.
    # 문자열로 만들 예정이므로, 첫 번째 숫자를 문자열로 변환하여 추가합니다.
    result_list = [str(num_list[0])]

    # 두 번째 숫자부터 마지막 숫자까지 이전 숫자와 비교하며 반복합니다.
    for i in range(len(num_list) - 1):
        current_num = num_list[i]
        next_num = num_list[i+1]

        # 2. 짝수 - 짝수 조건
        if current_num % 2 == 0 and next_num % 2 == 0:
            result_list.append("*") # 대시를 추가합니다.
            result_list.append(str(next_num)) # 다음 숫자를 문자열로 추가합니다.
        # 3. 홀수 - 홀수 조건
        elif current_num % 2 == 1 and next_num % 2 == 1:
            result_list.append("-") # 별표를 추가합니다.
            result_list.append(str(next_num)) # 다음 숫자를 문자열로 추가합니다.
        # 4. 그 외의 경우 (홀수-짝수 또는 짝수-홀수)
        else:
            result_list.append(str(next_num)) # 다음 숫자만 문자열로 추가합니다.

    # 리스트의 모든 요소를 하나의 문자열로 합쳐서 반환합니다.
    return "".join(result_list)

def Duplicate_Numbers(data_string):
    """
    공백으로 구분된 여러 숫자 문자열을 입력받아,
    각 문자열이 '0부터 9까지의 모든 숫자를 중복 없이 한 번씩 포함하는지' 확인하고
    그 결과를 반환합니다. (조건을 만족하면 True, 아니면 False)
    """
    # 1. 입력 문자열을 공백을 기준으로 분리하여 개별 숫자 문자열 리스트를 만듭니다.
    number_strings = data_string.split(' ')
    
    # 2. 리스트 컴프리헨션을 사용하여 각 숫자 문자열의 결과를 바로 생성합니다.
    # 각 num_str에 대해 (길이가 10이면서 동시에 중복이 없는지) 여부를 판단하여 True/False를 results 리스트에 추가합니다.
    results = [
        (len(num_str) == 10 and len(num_str) == len(set(num_str)))
        for num_str in number_strings
    ]
            
    # 3. 결과 출력
    for result in results:
        print(result, end=' ')
    print()
            
    # 4. 최종 결과 리스트를 반환합니다.
    return results

음, 신욱님께서 지금 느끼시는 감정은 아마도 많은 분들이 알고리즘 문제를 풀면서 한 번쯤은 경험하는 감정일 거예요. "귀찮음"이나 "생각을 쥐어짜야 하는 고통"은 단순히 신욱님만이 겪는 특별한 경험이 아니랍니다. 오히려 그것은 **성장통**과 같은 것이라고 생각하시면 좋을 것 같아요. 😌

몇 가지 관점에서 신욱님의 질문에 대해 생각해 볼 수 있습니다.

### 1. 알고리즘 문제풀이의 본질적인 어려움

*   **정답이 아닌 '최적'의 해법 찾기:** 알고리즘 문제는 단순히 정답을 찾아내는 것을 넘어, 가장 효율적이고 우아한 해법을 찾는 것이 목표일 때가 많습니다. 이는 기존에 알던 지식을 조합하고, 새로운 아이디어를 떠올리며, 여러 경우의 수를 탐색하는 **창의적 사고 과정**을 요구합니다. 이 과정이 에너지를 많이 소모하죠.
*   **새로운 시각과 논리적 사고:** 우리가 일상적으로 생각하는 방식과는 다른, 컴퓨터적인 논리 흐름으로 문제를 분석하고 해결해야 합니다. 처음에는 익숙하지 않아서 어렵고 답답하게 느껴질 수 있어요.
*   **복잡한 사고 과정:** 단순한 계산이 아니라, 데이터를 어떻게 구조화할지, 어떤 연산을 반복할지, 예외 상황은 어떻게 처리할지 등 여러 단계의 사고를 동시에 진행해야 합니다. 이로 인해 인지적 부하가 크게 발생합니다.

### 2. "귀찮음"의 원인

*   **인지적 부하 (Cognitive Load):** 머릿속에서 여러 가지 아이디어를 동시에 시뮬레이션하고, 오류를 찾아내고, 더 나은 방법을 찾는 과정은 매우 힘든 정신 활동입니다. 뇌가 평소보다 훨씬 많은 에너지를 사용하기 때문에 '귀찮다'는 감정으로 느껴질 수 있어요.
*   **좌절감과 두려움:** 막히는 지점에서 해답을 찾지 못하거나, 어렵게 생각해낸 방법이 틀렸을 때 오는 좌절감은 자연스럽게 '이거 왜 해야 하지?'라는 귀찮음으로 이어질 수 있습니다. 문제가 풀리지 않을 것 같은 두려움도 작용하겠죠.
*   **노력 대비 성과 부족:** 초반에는 많은 시간을 투자해도 문제가 쉽게 풀리지 않는 경우가 많습니다. 투입하는 노력에 비해 가시적인 성과가 적다고 느껴질 때도 귀찮음을 느낄 수 있습니다.

### 3. 이것이 '성장'의 과정이라는 증거

하지만 이러한 '귀찮음'과 '쥐어짜는' 과정은 매우 중요합니다.

*   **문제 해결 능력 향상:** 문제를 해결하기 위해 고뇌하는 과정 그 자체가 문제 해결 능력을 비약적으로 향상시킵니다. 쉽고 편하게 풀리는 문제는 이미 아는 것에 대한 반복일 뿐, 새로운 것을 배우는 데는 한계가 있습니다.
*   **논리력과 사고력 강화:** 막히는 부분에서 논리적인 오류를 찾고, 효율적인 방법을 모색하며, 다양한 경우의 수를 고려하는 훈련은 논리적 사고력을 극대화시킵니다.
*   **성취감:** 어렵게 푼 문제를 성공적으로 해결했을 때의 성취감은 그 어떤 것과도 바꿀 수 없을 정도로 값집니다. 그 성취감이 다음 문제를 풀 원동력이 되어줄 거예요.
*   **인내력과 끈기:** 어려운 문제 앞에서 포기하지 않고 끈기 있게 매달리는 경험은 인내심을 길러줍니다.
