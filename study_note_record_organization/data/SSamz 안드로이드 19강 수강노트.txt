package com.example.chapter19.Location
import android.Manifest
import android.content.pm.PackageManager
import android.location.LocationManager
import androidx.appcompat.app.AppCompatActivity
import android.os.Bundle
import android.util.Log
import android.widget.Toast
import androidx.activity.result.contract.ActivityResultContracts
import androidx.core.app.ActivityCompat
import androidx.core.content.ContextCompat
import com.example.chapter19.databinding.ActivityLocationPraBinding
class LocationPra : AppCompatActivity() {
lateinit var manager: LocationManager
fun getLocation(){
val location = manager.getLastKnownLocation(LocationManager.GPS_PROVIDER)
location?.let {
val latitude = location.latitude
val longitude = location.longitude
val accuracy = location.accuracy
Log.d("fifo23", "$latitude,$longitude,$accuracy")
}
Log.d("fifo23", "sdgsdg")
}
override fun onCreate(savedInstanceState: Bundle?) {
super.onCreate(savedInstanceState)
val binding = ActivityLocationPraBinding.inflate(layoutInflater)
setContentView(binding.root)
manager = getSystemService(LOCATION_SERVICE) as LocationManager
val permissionLauncher = registerForActivityResult(
ActivityResultContracts.RequestPermission()
){
if(it){
getLocation()
}
Toast.makeText(this, "denied", Toast.LENGTH_SHORT).show()
}
if(ContextCompat.checkSelfPermission(this,"android.permission.ACCESS_FINE_LOCATION") == PackageManager.PERMISSION_GRANTED){
getLocation()
}else{
permissionLauncher.launch("android.permission.ACCESS_FINE_LOCATION")
}

// 플레이서비스 위치서비스 이용 가능 할때 , 이용 불가능 할때, 지도이용 가능 할때
class MainActivity : AppCompatActivity(), GoogleApiClient.ConnectionCallbacks,
GoogleApiClient.OnConnectionFailedListener, OnMapReadyCallback{
//위치정보 프로바이더가 가능할때 얻기가 가능할때 사용할
lateinit var providerClient: FusedLocationProviderClient
//위치정보 얻기위한 판단
lateinit var apiClient: GoogleApiClient
//지도객체
var googleMap: GoogleMap? = null
override fun onCreate(savedInstanceState: Bundle?) {
super.onCreate(savedInstanceState)
setContentView(R.layout.activity_main)
//registerForActivityResult 인텐트를 실행시킬수 있는 코드를 작성했다면 이번엔
//권한 설정을 위한 코트로 작성
val requestPermissionLaungher = registerForActivityResult(
ActivityResultContracts.RequestMultiplePermissions()
){
if(it.all { permission -> permission.value == true }){
apiClient.connect()
}else{
Toast.makeText(this,"권한거부....",Toast.LENGTH_SHORT).show()
}
}
(supportFragmentManager.findFragmentById(R.id.mapView) as SupportMapFragment)!!.getMapAsync(this)
//클라이언트 api객체를 초기화 어떤 API를 쓸지 위치값과
providerClient = LocationServices.getFusedLocationProviderClient(this)
apiClient = GoogleApiClient.Builder(this)
.addApi(LocationServices.API)
.addConnectionCallbacks(this)
.addOnConnectionFailedListener(this)
.build()
//퍼미션부여되어있다면 컨넥트
if(ContextCompat.checkSelfPermission(this,Manifest.permission.ACCESS_FINE_LOCATION) !== PackageManager.PERMISSION_GRANTED ||
ContextCompat.checkSelfPermission(this, Manifest.permission.WRITE_EXTERNAL_STORAGE) !== PackageManager.PERMISSION_GRANTED ||
ContextCompat.checkSelfPermission(this,Manifest.permission.ACCESS_NETWORK_STATE) !== PackageManager.PERMISSION_GRANTED) {
requestPermissionLaungher.launch(
arrayOf(
Manifest.permission.ACCESS_FINE_LOCATION,
Manifest.permission.WRITE_EXTERNAL_STORAGE,
Manifest.permission.ACCESS_NETWORK_STATE
)
)
}else{
apiClient.connect()
}
}
//맵 좌표 위도 경도를 받아서 카메라 위치를 옮겨주고 마커로 표시해주는 함수구현
private fun moveMap(latitude: Double, logitude: Double){
val latlng = LatLng(latitude, logitude)
val position: CameraPosition = CameraPosition.Builder()
.target(latlng)
.zoom(16f)
.build()
//카메라 위치를 카메라 업데이트 펙토리를 사용해서 옮기자.
googleMap!!.moveCamera(CameraUpdateFactory.newCameraPosition(position))
//마커 아이콘 설정과 위치값지정 타티틀 지정
val markerOption = MarkerOptions()
markerOption.icon(BitmapDescriptorFactory.defaultMarker(BitmapDescriptorFactory.HUE_AZURE))
markerOption.position(latlng)
markerOption.title("MyLocation")
//마커를 추가하자.
googleMap?.addMarker(markerOption)
}
//위치 정보가 넘어오면 이거가지고 작업을 해주면된다.
override fun onConnected(p0: Bundle?) {
if(ContextCompat.checkSelfPermission(this, Manifest.permission.ACCESS_FINE_LOCATION)
=== PackageManager.PERMISSION_GRANTED){
providerClient.lastLocation.addOnSuccessListener (
this@MainActivity,
object: OnSuccessListener<Location>{
override fun onSuccess(p0: Location?) {
p0?.let {
val latitude = p0.latitude
val logitude = p0.longitude
moveMap(latitude,logitude)
}
}
})
apiClient.disconnect()
}
}
//맴버변수가 가용됬을쌔 사용되는 함수
override fun onMapReady(p0: GoogleMap?) {
googleMap = p0
}
//현재 가용할 수 있는 사용자가 없을때 호출되는 함수
override fun onConnectionFailed(p0: ConnectionResult) {
Log.d("fifo23", "map user failed....")
}

override fun onConnectionSuspended(p0: Int) {
}
