1) 익스프레스의 기능

라우팅 : 요청 URL에 따라 IF문이나 SWITCH문으로 라우팅 해야했지만 익스프레소 간편하게 사용할 수 있게되었다.
미들웨어 : 요청과 응답사이에 여러가지 기능을 수행함, 사용자들이 많아 패키지로 제공되어 만들지 않아도 된다.
템플릿 엔진 : 동적인 화면을 구성하는데 뷰를 사용함 이때 서버와 동적인 HTML페이지를 구성하는데 사용됨
정적인 파일 지원 : CSS파일이나 JS파일, 이미지 처럼 정적인 파일을 서비스할 수 있는 기능을 제공함 

자바스크립트를 사용해서 서버를 개발할때 적합한 익스프레스, 파이썬을 사용해서 서버를 개발할때 사용되는 장고나 플라스크, 루비언어를 사용하는 루비온 레일즈, PHP언어를 사용하는 라라 벨 등 다양한 프레임 워크가 존재한다.

익스프레스에서 서버와 클라이언트간에 통신할때 사용되는 요청 객체와 응답객체

req
req.body, req.bookies, req.headers, req.params, req.query

res
res.download, res.end, res.json, res.jsonp, res.redirect, res.render, res.sender, res.sednFile, res.sendStatus, res.status

노드에서 주로 사용되는 익스프레스
라우트 파라미터는 URL에 함게 담아서 요청하는 값을 라우트 파라미터라고 하고 /contacts 뒤에 /contacts::id 추가해서 사용한다.


const express = require("express");
const app = express();
const port = 3000;
// const path = require("path");

app.get("/", (req, res) => {
  res.status(200);
  res.send("Hello, World!");
  // res.json({ message: "Hello, world!"});
});

// 모든 연락처 가져오기
app.get("/contacts", (req, res) => {
  res.status(200).send("Contacts Page");
  // res.sendFile(__dirname + "/assets/contacts.html");
});

// 새 연락처 추가하기
app.post("/contacts", (req, res) => {
  res.status(201).send("Create Contacts");
});

// 연락처 상세 보기
app.get("/contacts/:id", (req, res) => {
  res.status(200).send(`View Contact for ID: ${req.params.id}`);
});

// 연락처 수정하기
app.put("/contacts/:id", (req, res) => {
  res.status(200).send(`Update Contact for ID: ${req.params.id}`);
});

// 연락처 삭제하기
app.delete("/contacts/:id", (req, res) => {
  res.status(200).send(`Delete Contactfor ID: ${req.params.id}`);
});


app.listen(port, () => {
  console.log(`${port}번 포트에서 서버 실행 중`);
});
