onStartJop() 리턴값
(false: 벡그라운드 작업이 완벽히 종료 되었다.
true: 벡그라운드 작업이 아직 완전히 종료되지 않았다.)
onStartJob() 함수에서 false 리던
*onStartjab() 에서 시간이 오래걸리는 작업을 Thread로 하고있다면 true 를 리턴해서 잠시 기다렸다가 실행
onStopjab() 은 비정상 종료될때만 실행 된다. 조건이 바꼇을때 혹은 외부에서 캔슬시켰을때

package com.example.chapter15.JabService
import android.app.job.JobInfo
import android.app.job.JobScheduler
import android.content.ComponentName
import android.net.Uri
import android.os.Build
import androidx.appcompat.app.AppCompatActivity
import android.os.Bundle
import android.os.PersistableBundle
import com.example.chapter15.databinding.ActivityJabServicePraBinding
class JabServicePra : AppCompatActivity() {
override fun onCreate(savedInstanceState: Bundle?) {
super.onCreate(savedInstanceState)
val binding = ActivityJabServicePraBinding.inflate(layoutInflater)
setContentView(binding.root)
binding.jobServiceStartBtn.setOnClickListener {
//시스템에 요청하기 위해 잡스케쥴라변수와 데이터를 넣을 extras 변수 생성
val scheduler = getSystemService(JOB_SCHEDULER_SERVICE) as JobScheduler
val extras = PersistableBundle()
//번들에 데이타를 담는 작업
extras.putString("extra_data", "hello fifo23")
//빌더로 시스템의 조건을 JobInfo에 네트워크 가 끊겼을때의 상황을 담고 데이터를 잠고 빌더를 빌드하고
// 시스템에 의뢰해서 잡스케쥴러에 등록실행
val builder = JobInfo.Builder(1, ComponentName(this,JobService::class.java))
builder.setRequiredNetworkType(JobInfo.NETWORK_TYPE_UNMETERED)
builder.setExtras(extras)
scheduler.schedule(builder.build())
}
}
}
